class Solution {
  public:
    int minimumCost(vector<int> &cost, int w) {
        // code here
        
        // Unbounded Knapsack DP problem
        int INF= 1e9;
        vector <int> dp(w+1,INF);
        dp[0]=0;
        for(int i =1;i<=w;i++){
            for(int j =0;j<cost.size();j++){
                if(cost[j]==-1){
                    continue;
                }
                if(i>=j+1){
                    dp[i]= min(dp[i],dp[i-j-1]+cost[j]);
                }
            }
        }
        return dp[w]== INF? -1:dp[w];
        
        
    }
};
